## Reproduction commands

Run from the repository root:

```sh
# Establish the product-source inventory used by the following commands.
files="$(
  {
    git ls-files '*.mbt'
    git ls-files --others --exclude-standard '*.mbt'
  } |
    sort -u |
    grep -Ev '^(vendor/|ui/rabbita-desk/\.mooncakes/|ui/rabbita-desk/_build/|_build/|target/)'
)"

# Implementation/test file and line counts.
printf '%s\n' "$files" |
  awk '/(_test|_wbtest)\.mbt$/{test++; next} {impl++} END{print impl, test}'
printf '%s\n' "$files" |
  grep -Ev '(_test|_wbtest)\.mbt$' |
  xargs wc -l |
  tail -n 1
printf '%s\n' "$files" |
  grep -E '(_test|_wbtest)\.mbt$' |
  xargs wc -l |
  tail -n 1

# Largest implementation and test files by line count.
printf '%s\n' "$files" |
  grep -Ev '(_test|_wbtest)\.mbt$' |
  xargs wc -l |
  grep -v ' total$' |
  sort -nr |
  head -n 10

  grep -E '(_test|_wbtest)\.mbt$' |
  xargs wc -l |
  grep -v ' total$' |
```
